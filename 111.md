可以。你的业务链路直接写成：

原始 ID：1
    ↓ 转成固定 8 位 Base36

00000001

二、完整 Java 17 实现

下面这份代码实现：

encode(1)              // 1 -> 固定8位订单号
decode("1IRPSXFF")     // 固定8位订单号 -> 1
import java.util.Locale;
import java.util.Objects;
import org.bouncycastle.crypto.fpe.FPEFF1Engine;
import org.bouncycastle.crypto.params.FPEParameters;
import org.bouncycastle.crypto.params.KeyParameter;
/**
 * 将递增数字 ID 转成固定 8 位公开订单号，并支持逆向还原。
 *
 * 示例：
 * 1 -> 1IRPSXFF
 * 1IRPSXFF -> 1
 *
 * 注意：
 * 具体输出由 AES key 和 tweak 决定。
 */
public final class OrderNoCodec {
    /**
     * 订单号允许使用的字符。
     *
     * Base36：
     * 0-9 + A-Z
     */
    private static final String ALPHABET =
            "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    /**
     * Base36。
     */
    private static final int RADIX = ALPHABET.length();
    /**
     * 固定输出 8 位。
     */
    private static final int LENGTH = 8;
    /**
     * 36^8。
     *
     * 最大支持：
     * 1 ～ 2,821,109,907,455
     */
    private static final long CAPACITY =
            2_821_109_907_456L;
    private final byte[] aesKey;
    private final byte[] tweak;
    /**
     * @param aesKey AES 密钥，只能是 16、24 或 32 字节
     * @param tweak  固定业务标识，例如 ORDER-NO-V1
     */
    public OrderNoCodec(byte[] aesKey, byte[] tweak) {
        Objects.requireNonNull(aesKey, "aesKey");
        Objects.requireNonNull(tweak, "tweak");
        if (aesKey.length != 16
                && aesKey.length != 24
                && aesKey.length != 32) {
            throw new IllegalArgumentException(
                    "AES key must be 16, 24 or 32 bytes"
            );
        }
        this.aesKey = aesKey.clone();
        this.tweak = tweak.clone();
    }
    /**
     * 正向转换：
     *
     * 数字 ID -> 固定 8 位订单号。
     *
     * @param id 从 1 开始的数字 ID
     * @return 固定 8 位订单号
     */
    public String encode(long id) {
        validateId(id);
        /*
         * 1 -> "1"
         * 35 -> "Z"
         * 36 -> "10"
         */
        String base36 = Long.toString(id, RADIX)
                .toUpperCase(Locale.ROOT);
        /*
         * 补齐到 8 位：
         *
         * 1  -> 00000001
         * 36 -> 00000010
         */
        String fixedWidth = "0".repeat(
                LENGTH - base36.length()
        ) + base36;
        /*
         * 00000001 -> FF1 -> 例如 1IRPSXFF
         */
        return transform(fixedWidth, true);
    }
    /**
     * 逆向转换：
     *
     * 固定 8 位订单号 -> 原始数字 ID。
     *
     * 支持：
     * ABCD1234
     * abcd1234
     * ABCD-1234
     * ABCD 1234
     */
    public long decode(String orderNo) {
        String normalized = normalize(orderNo);
        /*
         * 例如：
         * 1IRPSXFF -> FF1解密 -> 00000001
         */
        String plainText = transform(
                normalized,
                false
        );
        /*
         * 00000001 -> Base36解析 -> 1
         */
        long id = Long.parseLong(
                plainText,
                RADIX
        );
        validateId(id);
        return id;
    }
    /**
     * 标准化用户输入。
     *
     * abcd-1234 -> ABCD1234
     */
    public String normalize(String orderNo) {
        Objects.requireNonNull(
                orderNo,
                "orderNo"
        );
        String normalized = orderNo
                .trim()
                .replace("-", "")
                .replace(" ", "")
                .toUpperCase(Locale.ROOT);
        if (normalized.length() != LENGTH) {
            throw new IllegalArgumentException(
                    "订单号必须是 8 位字符"
            );
        }
        // 提前校验是否包含非法字符。
        toDigits(normalized);
        return normalized;
    }
    /**
     * 前端展示格式：
     *
     * ABCD1234 -> ABCD-1234
     */
    public String formatForDisplay(String orderNo) {
        String normalized = normalize(orderNo);
        return normalized.substring(0, 4)
                + "-"
                + normalized.substring(4);
    }
    /**
     * 执行 FF1 加密或者解密。
     */
    private String transform(
            String value,
            boolean encrypt
    ) {
        byte[] input = toDigits(value);
        byte[] output = new byte[input.length];
        /*
         * 每次操作创建一个新的 Engine。
         *
         * 不要把 FPEFF1Engine 作为多线程共享单例，
         * 因为它需要通过 init() 设置加密或解密状态。
         */
        FPEFF1Engine engine =
                new FPEFF1Engine();
        FPEParameters parameters =
                new FPEParameters(
                        new KeyParameter(aesKey),
                        RADIX,
                        tweak,
                        false
                );
        /*
         * true  = 加密
         * false = 解密
         */
        engine.init(
                encrypt,
                parameters
        );
        int written = engine.processBlock(
                input,
                0,
                input.length,
                output,
                0
        );
        if (written != input.length) {
            throw new IllegalStateException(
                    "Unexpected FF1 output length: "
                            + written
            );
        }
        return fromDigits(output);
    }
    /**
     * 把字符串字符转换为 FF1 数字数组。
     *
     * 0 -> 0
     * 9 -> 9
     * A -> 10
     * Z -> 35
     */
    private static byte[] toDigits(String value) {
        byte[] result =
                new byte[value.length()];
        for (int i = 0; i < value.length(); i++) {
            char character = value.charAt(i);
            int digit =
                    ALPHABET.indexOf(character);
            if (digit < 0) {
                throw new IllegalArgumentException(
                        "订单号包含非法字符: "
                                + character
                );
            }
            result[i] = (byte) digit;
        }
        return result;
    }
    /**
     * 把 FF1 数字数组转换回订单号字符串。
     */
    private static String fromDigits(
            byte[] digits
    ) {
        StringBuilder result =
                new StringBuilder(digits.length);
        for (byte rawDigit : digits) {
            int digit =
                    Byte.toUnsignedInt(rawDigit);
            if (digit >= RADIX) {
                throw new IllegalStateException(
                        "Invalid FF1 digit: "
                                + digit
                );
            }
            result.append(
                    ALPHABET.charAt(digit)
            );
        }
        return result.toString();
    }
    private static void validateId(long id) {
        if (id <= 0 || id >= CAPACITY) {
            throw new IllegalArgumentException(
                    "ID 必须处于 1 到 "
                            + (CAPACITY - 1)
                            + " 之间"
            );
        }
    }
}

FPEParameters 官方提供了 key、radix、tweak 和 useInverse 参数，FPEFF1Engine.init(true/false, parameters) 分别初始化加密和解密模式。

代码中最后一个参数明确使用：

false

即不启用 inverse AES function，这也符合 NIST 最新 FF1 修订草案中的方向。

三、调用示例

下面是一个可以直接运行的测试。

import java.nio.charset.StandardCharsets;
import java.util.HexFormat;
public class OrderNoDemo {
    public static void main(String[] args) {
        /*
         * 演示密钥。
         *
         * 生产环境不要硬编码。
         * 应当从 KMS、Vault、环境变量或 Secret 中读取。
         */
        byte[] aesKey = HexFormat.of().parseHex(
                "00112233445566778899AABBCCDDEEFF"
              + "000102030405060708090A0B0C0D0E0F"
        );
        /*
         * tweak 必须长期固定。
         */
        byte[] tweak = "ORDER-NO-V1"
                .getBytes(StandardCharsets.US_ASCII);
        OrderNoCodec codec =
                new OrderNoCodec(
                        aesKey,
                        tweak
                );
        long originalId = 1L;
        /*
         * 1 -> 固定8位订单号
         */
        String orderNo =
                codec.encode(originalId);
        /*
         * 固定8位订单号 -> 1
         */
        long restoredId =
                codec.decode(orderNo);
        System.out.println(
                "原始 ID: " + originalId
        );
        System.out.println(
                "订单号: " + orderNo
        );
        System.out.println(
                "显示格式: "
                        + codec.formatForDisplay(orderNo)
        );
        System.out.println(
                "逆向 ID: " + restoredId
        );
        if (originalId != restoredId) {
            throw new AssertionError(
                    "FF1 正反转换失败"
            );
        }
    }
}

四、Spring Boot 中怎么使用

最简单的方式是注册为 Spring Bean。

配置类

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
@Configuration
public class OrderNoConfiguration {
    @Bean
    public OrderNoCodec orderNoCodec(
            @Value("${order-number.key-base64}")
            String keyBase64
    ) {
        byte[] aesKey = Base64
                .getDecoder()
                .decode(keyBase64);
        byte[] tweak = "ORDER-NO-V1"
                .getBytes(StandardCharsets.US_ASCII);
        return new OrderNoCodec(
                aesKey,
                tweak
        );
    }
}

application.yml

order-number:
  key-base64: ${ORDER_NUMBER_KEY_BASE64}

环境变量中设置一个长期固定的 32 字节密钥：

ORDER_NUMBER_KEY_BASE64=你的Base64密钥

绝对不能每次启动时随机生成新密钥。

否则就会变成：

今天：
1 -> ABCD1234
重启并换密钥后：
1 -> QWER5678

以前发给用户的订单号将无法解密。

六、用户用 8 位订单号查询

用户输入：

1IRP-SXFF

后端处理：

public OrderResponse findByOrderNo(
        String inputOrderNo
) {
    /*
     * 1IRP-SXFF
     *      ↓
     * 1IRPSXFF
     *      ↓ FF1解密
     * 1
     */
    long id =
            orderNoCodec.decode(inputOrderNo);
    Order order = orderRepository
            .findById(id)
            .orElseThrow(
                    () -> new IllegalArgumentException(
                            "订单不存在"
                    )
            );
    return toResponse(order);
}

完整逻辑是：

用户输入8位订单号
        ↓
标准化大小写、去掉横线和空格
        ↓
FF1解密
        ↓
得到原始数字ID
        ↓
按数据库主键查询

这样可以直接走主键：

SELECT *
FROM orders
WHERE id = ?;

查询效率非常高，而且不一定需要额外保存 order_no 字段。

七、多租户查询时必须带权限条件

不要只执行：

orderRepository.findById(id)

更稳妥的是：

orderRepository.findByIdAndTenantId(
        id,
        currentTenantId
)

例如：

public OrderResponse findByOrderNo(
        String inputOrderNo,
        Long currentTenantId
) {
    long id =
            orderNoCodec.decode(inputOrderNo);
    Order order = orderRepository
            .findByIdAndTenantId(
                    id,
                    currentTenantId
            )
            .orElseThrow(
                    () -> new IllegalArgumentException(
                            "订单不存在"
                    )
            );
    return toResponse(order);
}

订单号只是公开标识，不是权限凭证。

import org.junit.jupiter.api.BeforeEach;

class OrderNoCodecTest {

    @Test

        String orderNo =
                codec.encode(1L);
        assertEquals(
                8,

        );
        assertEquals(
                1L,

        );
    }
    @Test
    void shouldSupportFormattedInput() {
        String orderNo =
                codec.encode(1L);
        String formatted =
                codec.formatForDisplay(orderNo);
        assertEquals(
                1L,
                codec.decode(formatted)
        );
    }
    @Test
    void shouldBeDeterministic() {
        String first =
                codec.encode(12345L);
        String second =
                codec.encode(12345L);
        assertEquals(
                first,
                second
        );
    }
    @Test
    void shouldNotRepeatForDifferentIds() {
        Set<String> orderNumbers =
                new HashSet<>();
        for (long id = 1; id <= 10_000; id++) {
            String orderNo =
                    codec.encode(id);
            assertTrue(
                    orderNumbers.add(orderNo),
                    "发现重复订单号，ID=" + id
            );
            assertEquals(
                    id,
                    codec.decode(orderNo)
            );
        }
    }
    @Test
    void shouldRejectInvalidId() {
        assertThrows(
                IllegalArgumentException.class,
                () -> codec.encode(0)
        );
        assertThrows(
                IllegalArgumentException.class,
                () -> codec.encode(-1)
        );
    }
    @Test
    void shouldRejectInvalidOrderNo() {
        assertThrows(
                IllegalArgumentException.class,
                () -> codec.decode("ABC")
        );
        assertThrows(

九、为什么不同 ID 不会重复

FF1 不是：

随机生成8位字符串

而是：

在全部8位字符空间中做一一置换

因此在同一套配置下：

输入1 ≠ 输入2

就一定有：

encode(1) ≠ encode(2)

前提是以下配置完全一致：

AES key 一致
tweak 一致
字符集一致
长度一致
FF1配置一致

8 位 Base36 总容量是：

36^8
= 2,821,109,907,456

因此可用 ID 范围是：

1 ～ 2,821,109,907,455

⸻

十、只允许纯数字时怎么改

上面的方案输出：

数字 + 大写字母

例如：

1IRPSXFF

如果业务要求严格的 8 位纯数字，只需要改三个地方：

private static final String ALPHABET =
        "0123456789";
private static final int RADIX = 10;
private static final long CAPACITY =
        100_000_000L;

此时支持范围只有：

1 ～ 99,999,999

并且字段必须使用：

CHAR(8)

不要使用：

INT
BIGINT

因为 FF1 输出可能以 0 开头：

00127364

存成数字后会变成：

127364

固定 8 位格式就丢失了。

⸻

最终落地方式

你现在可以直接删除这条旧链路：

1
↓
UUID
↓
截取8位

改成：

数据库 ID：1
       ↓
orderNoCodec.encode(1)
       ↓
固定8位订单号
       ↓
展示给用户

用户查询时：

固定8位订单号
       ↓
orderNoCodec.decode(orderNo)
       ↓
恢复数据库 ID：1
       ↓
按 id + tenantId 查询

对于你现在“从 1 开始的数字 ID”，这是最直接、固定 8 位、可逆且不会因截断而碰撞的实现。
